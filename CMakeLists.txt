cmake_minimum_required(VERSION 3.20)
project(MiniDatabaseEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(minidb src/Database.cpp src/Table.cpp src/Parser.cpp src/Storage.cpp src/BPlusTree.cpp)
target_include_directories(minidb PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(minidb_cli src/main.cpp)
target_link_libraries(minidb_cli PRIVATE minidb)

include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(minidb_tests tests/ParserTest.cpp tests/StorageTest.cpp tests/DatabaseTest.cpp)
  target_link_libraries(minidb_tests PRIVATE minidb GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(minidb_tests)
endif()
